### [OK] (n) THE ADMISSION COMPARISON: all four refuse-directions + the legal admit  (container=main role=sm_dp_runtime db=smdb)
--- SQL ---
WITH wm AS (SELECT s.tenant_id,s.instance_id,s.instance_watermark,
      (SELECT max(generation) FROM sm_data.placement_watermark p WHERE p.tenant_id=s.tenant_id AND p.instance_id=s.instance_id) AS any_p,
      (SELECT generation FROM sm_data.placement_watermark p WHERE p.tenant_id=s.tenant_id AND p.instance_id=s.instance_id AND p.placement_id='b1b1b1b1-1111-4111-8111-b1b1b1b1b1b1') AS own_p1
    FROM sm_data.apply_scope s WHERE s.tenant_id='44444444-4444-4444-8444-444444444444' AND s.instance_id='a1a1a1a1-1111-4111-8111-a1a1a1a1a1a1'),
cases(direction,scope,g,bar) AS (
  SELECT '1 placement-scoped apply <= INSTANCE watermark','placement P2',10::bigint,
    GREATEST(w.instance_watermark, coalesce((SELECT generation FROM sm_data.placement_watermark WHERE tenant_id='44444444-4444-4444-8444-444444444444' AND instance_id='a1a1a1a1-1111-4111-8111-a1a1a1a1a1a1' AND placement_id='b2b2b2b2-2222-4222-8222-b2b2b2b2b2b2'),0)) FROM wm w
  UNION ALL SELECT '2 placement-scoped apply <= its OWN placement watermark','placement P1',15, GREATEST(w.instance_watermark, coalesce(w.own_p1,0)) FROM wm w
  UNION ALL SELECT '3 instance-wide apply <= INSTANCE watermark','instance-wide',10, GREATEST(w.instance_watermark,0) FROM wm w
  UNION ALL SELECT '4 instance-wide apply <= ANY placement watermark','instance-wide',15, GREATEST(w.instance_watermark, coalesce(w.any_p,0)) FROM wm w
  UNION ALL SELECT '5 LEGAL: G strictly above every bar','instance-wide',21, GREATEST(w.instance_watermark, coalesce(w.any_p,0)) FROM wm w)
SELECT direction, scope, g AS "G", bar, CASE WHEN g <= bar THEN 'REFUSE' ELSE 'ADMIT' END AS verdict FROM cases ORDER BY 1;
